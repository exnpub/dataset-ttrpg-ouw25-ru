   cp .env.example .env
   # Добавить HF_TOKEN в .env

2️⃣  СОЗДАНИЕ ДАТАСЕТА (10 минут)
   python3 scripts/prepare_for_huggingface.py
   # Генерирует:
   #   • Парке-файлы для HF
   #   • dataset_info.yaml
   #   • Статистику

3️⃣  ЛОКАЛЬНОЕ ТЕСТИРОВАНИЕ (5 минут)
   python3 scripts/test_hf_dataset.py
   # Проверяет:
   #   • Формат данных
   #   • Все конфигурации
   #   • Целостность

4️⃣  ЗАГРУЗКА НА HF HUB (2 минуты)
   python3 scripts/upload_to_huggingface.py
   # Выгружает:
   #   • Все данные
   #   • Метаданные
   #   • Dataset Card

5️⃣  ФИНАЛЬНАЯ ПРОВЕРКА (2 минуты)
   # На HF Hub: https://huggingface.co/datasets/exnihilum/...
   python3 -c "from datasets import load_dataset; ds = load_dataset('exnihilum/...'); print(ds)"

⏱️  ИТОГО: ~25 минут (+ время загрузки на HF)
